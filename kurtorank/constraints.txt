# constraints.txt for kurtorank
#
# kurtorank shares a conda env with wsinsight, sptxinsight, hplot and
# wsinsight-train. These pins mirror wsinsight/constraints.txt and
# sptxinsight/constraints.txt so installing kurtorank cannot drift that env.
#
# ALWAYS install with this file:
#     pip install -c constraints.txt -e .
# or let bash ./conda-setup.sh do it, which additionally installs the deps
# explicitly and then adds kurtorank itself with --no-deps.
#
# kurtorank itself is pure PyTorch (no TensorFlow/Keras), but it shares numpy,
# anndata, dask and the CUDA wheels with packages that do use TensorFlow.

# --- the numpy<2 stack ----------------------------------------------------
numpy<2.0.0
# setuptools>=81 removes pkg_resources. stardist 0.9.2 no longer imports it, but
# other packages in the shared env still do (imageio, tensorboard, werkzeug, ...).
setuptools<67.0.0
h5py==3.15.0

# --- AnnData / spatial stack ----------------------------------------------
# Must match wsinsight/sptxinsight: capping at <0.12 here silently DOWNGRADED a
# working env from 0.12.19 to 0.11.4, and 0.12 is required to read h5ad files
# written in the 0.12 format. Nothing in the dependency tree actually asks for
# <0.12 (scanpy >=0.8, squidpy >=0.9, spatialdata >=0.9.1, tiledbsoma >=0.10.1).
anndata>=0.11,<0.13
zarr==2.18.7                 # zarr 3.x requires numpy>=2
xarray<2025
scanpy<1.11            # last numpy<2-compatible release
squidpy<1.7
spatialdata>=0.4,<0.5
spatialdata-io>=0.6,<0.7

# --- dask / fsspec family -------------------------------------------------
# wsinsight needs dask<2024.11.0; this repo's pyproject allows <=2024.11.2.
# 2024.10.0 is the only version both accept.
dask==2024.10.0
dask-expr==1.1.16
distributed==2024.10.0
fsspec==2025.9.0
s3fs==2025.9.0
gcsfs==2025.9.0
aiobotocore==2.25.0
# Mirrors wsinsight, the reference set: aiobotocore 2.25.0 requires
# botocore<1.40.50, so the whole family stays on the 1.40.x generation.
boto3==1.40.49
botocore==1.40.49
# tiledbsoma / AnnData parquet I/O; matches wsinsight's pin.
pyarrow==21.0.0

# --- PyTorch --------------------------------------------------------------
# pyproject leaves torch unpinned. Unconstrained, pip resolves to torch>=2.9,
# which pulls the CUDA 13 wheels; those overwrite nvidia/cudnn/lib and break
# TensorFlow (StarDist) elsewhere in the shared env.
torch==2.8.0
torchvision==0.23.0
nvidia-cublas-cu12==12.8.4.1
nvidia-cuda-cupti-cu12==12.8.90
nvidia-cuda-nvrtc-cu12==12.8.93
nvidia-cuda-runtime-cu12==12.8.90
nvidia-cudnn-cu12==9.10.2.21
nvidia-cufft-cu12==11.3.3.83
nvidia-cufile-cu12==1.13.1.3
nvidia-curand-cu12==10.3.9.90
nvidia-cusolver-cu12==11.7.3.90
nvidia-cusparse-cu12==12.5.8.93
nvidia-cusparselt-cu12==0.7.1
nvidia-nccl-cu12==2.27.3
nvidia-nvjitlink-cu12==12.8.93
nvidia-nvtx-cu12==12.8.90

# --- remaining direct deps from pyproject.toml -----------------------------
# Same versions as the shared wsinsight env.
click==8.3.0
matplotlib==3.10.7
pandas==2.3.3
scipy==1.16.2
seaborn==0.13.2
